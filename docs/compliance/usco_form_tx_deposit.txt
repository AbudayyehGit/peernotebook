   122:   /**
   123:    * Dispatches transactional email via native fetch() to Brevo REST API v3
   124:    * Wraps requests in robust try/catch blocks with detailed logging.
   125:    */
   126:   public static async sendEmail(options: BrevoRestEmailOptions): Promise<BrevoRestDispatchResult> {
   127:     const apiKey = options.apiKey || this.getApiKey();
   128:     const sender = options.sender || this.DEFAULT_SENDER;
   129:     const replyTo = options.replyTo || this.DEFAULT_REPLY_TO;
   130:     const timestamp = new Date().toISOString();
   131: 
   132:     // Standardize recipient array
   133:     let recipients: BrevoRecipient[] = [];
   134:     if (typeof options.to === 'string') {
   135:       recipients = [{ email: options.to.trim() }];
   136:     } else if (Array.isArray(options.to)) {
   137:       recipients = options.to.map((item) => {
   138:         if (typeof item === 'string') return { email: item.trim() };
   139:         return item;
   140:       });
   141:     }
   142: 
   143:     const textContent = options.textContent || this.htmlToPlainText(options.htmlContent);
   144: 
   145:     const baseDetails = {
   146:       sender,
   147:       to: recipients,
   148:       subject: options.subject,
   149:       timestamp,
   150:       hasTextAlternative: Boolean(textContent && textContent.length > 0),
   151:       endpoint: this.ENDPOINT
   152:     };
   153: 
   154:     console.log(
   155:       `[BREVO_REST_INIT] Dispatching "${options.subject}" to ${recipients.map((r) => r.email).join(', ')} via Brevo REST API...`
   156:     );
   157: 
   158:     // If no API key is present and simulation fallback allowed
   159:     if (!apiKey || !apiKey.startsWith('xkeysib-')) {
   160:       if (options.allowSimulationFallback) {
   161:         const simId = `<sim_${Date.now()}_${Math.random().toString(36).substring(2, 8)}@brevo-rest.peernotebook.com>`;
   162:         console.warn('[BREVO_REST_SIMULATED] Valid API key missing; dispatched in simulation mode:', simId);
   163:         return {
   164:           success: true,
   165:           status: 'simulated',
   166:           provider: 'simulated-brevo',
   167:           messageId: simId,
   168:           details: baseDetails
   169:         };
   170:       }
   171: 
   172:       console.error('[BREVO_REST_ERROR] Missing or malformed BREVO_API_KEY (must start with xkeysib-)');
   173:       return {
   174:         success: false,
   175:         status: 'failed',
   176:         provider: 'brevo-rest-api',
   177:         error: {
   178:           code: 'INVALID_API_KEY',
   179:           message: 'BREVO_API_KEY is missing or invalid. Expected key starting with xkeysib-',
   180:           suggestion: 'Check environment variables or verify key at app.brevo.com/settings/keys/api'
   181:         },
   182:         details: baseDetails
   183:       };
   184:     }
   185: 
   186:     try {
   187:       // Brevo v3 REST API payload
   188:       const payload: Record<string, any> = {
   189:         sender: {
   190:           name: sender.name,
   191:           email: sender.email
   192:         },
   193:         to: recipients.map((r) => ({
   194:           email: r.email,
   195:           ...(r.name ? { name: r.name } : {})
   196:         })),
   197:         replyTo: {
   198:           name: replyTo.name,
   199:           email: replyTo.email
   200:         },
   201:         subject: options.subject,
   202:         htmlContent: options.htmlContent,
   203:         textContent,
   204:         headers: {
   205:           'X-Mailer': 'PeerNotebook-Brevo-REST-v3',
   206:           'List-Unsubscribe': `<mailto:${this.PRIMARY_SUPPORT_EMAIL}?subject=unsubscribe>`,
   207:           ...(options.headers || {})
   208:         }
   209:       };
   210: 
   211:       if (options.tags && options.tags.length > 0) {
   212:         payload.tags = options.tags;
   213:       }
   214:       if (options.cc && options.cc.length > 0) {
   215:         payload.cc = options.cc;
   216:       }
   217:       if (options.bcc && options.bcc.length > 0) {
   218:         payload.bcc = options.bcc;
   219:       }
   220: 
   221:       const headers = {
   222:         'accept': 'application/json',
   223:         'content-type': 'application/json',
   224:         'api-key': apiKey
   225:       };
   226: 
   227:       const response = await fetch(this.ENDPOINT, {
   228:         method: 'POST',
   229:         headers,
   230:         body: JSON.stringify(payload)
   231:       });
   232: 
   233:       const responseText = await response.text();
   234:       let responseJson: any = {};
   235:       try {
   236:         responseJson = JSON.parse(responseText);
   237:       } catch {
   238:         responseJson = { message: responseText };
   239:       }
   240: 
   241:       if (!response.ok) {
   242:         console.error(
   243:           `[BREVO_REST_FAILURE] HTTP ${response.status} from ${this.ENDPOINT}:`,
   244:           JSON.stringify(responseJson, null, 2)
   245:         );
   246: 
   247:         let suggestion = 'Review Brevo dashboard configuration.';
   248:         if (response.status === 401) {
   249:           suggestion = 'Authentication failed. Verify BREVO_API_KEY at app.brevo.com/settings/keys/api.';
   250:         } else if (response.status === 400) {
   251:           suggestion = `Payload error: ${responseJson.message || 'Check recipient formatting and HTML content.'}`;
   252:         } else if (response.status === 403) {
   253:           suggestion = `Sender domain or email "${sender.email}" is unverified in Brevo. Verify at app.brevo.com/senders.`;
   254:         } else if (response.status === 402) {
   255:           suggestion = 'Account quota exceeded (300 emails/day on free plan).';
   256:         }
   257: 
   258:         // Check if simulation fallback is requested on network/auth error
   259:         if (options.allowSimulationFallback) {
   260:           console.warn('[BREVO_REST_FALLBACK] Live API failed, falling back to simulated dispatch to avoid user-facing disruption.');
   261:           return {
   262:             success: true,
   263:             status: 'simulated',
   264:             provider: 'simulated-brevo',
   265:             messageId: `<sim_fallback_${Date.now()}@brevo.com>`,
   266:             error: {
   267:               statusCode: response.status,
   268:               code: responseJson.code || 'BREVO_API_ERROR',
   269:               message: responseJson.message || responseText,
   270:               details: responseJson,
   271:               suggestion
   272:             },
   273:             details: baseDetails
   274:           };
   275:         }
   276: 
   277:         return {
   278:           success: false,
   279:           status: 'failed',
   280:           provider: 'brevo-rest-api',
   281:           statusCode: response.status,
   282:           error: {
   283:             statusCode: response.status,
   284:             code: responseJson.code || `HTTP_${response.status}`,
   285:             message: responseJson.message || responseText || 'Unknown Brevo API error',
   286:             details: responseJson,
   287:             suggestion
   288:           },
   289:           details: baseDetails
   290:         };
   291:       }
   292: 
   293:       const messageId = responseJson.messageId || `<brevo_${Date.now()}@smtp-relay.brevo.com>`;
   294:       console.log(
   295:         `[BREVO_REST_SUCCESS] Successfully dispatched email via Brevo REST API. Message ID: ${messageId}`
   296:       );
   297: 
   298:       return {
   299:         success: true,
   300:         status: 'delivered',
   301:         provider: 'brevo-rest-api',
   302:         messageId,
   303:         statusCode: response.status,
   304:         details: baseDetails
   305:       };
   306:     } catch (networkErr: unknown) {
   307:       const errorMsg = networkErr instanceof Error ? networkErr.message : String(networkErr);
   308:       console.error('[BREVO_REST_NETWORK_ERROR] Exception during fetch() to Brevo API:', errorMsg);
   309: 
   310:       if (options.allowSimulationFallback) {
   311:         console.warn('[BREVO_REST_NETWORK_FALLBACK] Using simulation fallback for network exception.');
   312:         return {
   313:           success: true,
   314:           status: 'simulated',
   315:           provider: 'simulated-brevo',
   316:           messageId: `<sim_net_${Date.now()}@brevo.com>`,
   317:           error: {
   318:             code: 'NETWORK_EXCEPTION',
   319:             message: errorMsg,
   320:             suggestion: 'Verify outbound HTTPS connectivity to https://api.brevo.com'
   321:           },
   322:           details: baseDetails
   323:         };
   324:       }
   325: 
   326:       return {
   327:         success: false,
   328:         status: 'failed',
   329:         provider: 'brevo-rest-api',
   330:         error: {
   331:           code: 'NETWORK_EXCEPTION',
   332:           message: errorMsg,
   333:           suggestion: 'Verify outbound HTTPS connectivity to https://api.brevo.com'
   334:         },
   335:         details: baseDetails
   336:       };
   337:     }
   338:   }
